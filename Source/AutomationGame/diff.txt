+int32 UGoldbergGrid::ComputeFrequencyForTileSize(float TileSizeUU,
+                                                 float Radius) {
+  // On a Goldberg polyhedron of subdivision frequency f, each edge of the base
+  // icosahedron is split into f segments.  The icosahedron edge angular length
+  // on a unit sphere is: 2 * arctan(1 / phi) Γëê 1.10714 rad  (Γëê 63.43┬░)
+  // (phi = golden ratio = (1+sqrt(5))/2 Γëê 1.618)
+  // The edge length on the sphere in world units = Radius * angular_length.
+  // Each segment then has an arc-length of (Radius * angular_length) / f.
+  //
+  // IMPORTANT: TileSizeUU is the hex "radius" (center-to-vertex).
+  // The center-to-center distance (neighbor spacing) that the hex mesh
+  // expects is TileSizeUU * sqrt(3).  So we solve:
+  //   (Radius * angular_length) / f = TileSizeUU * sqrt(3)
+  //   => f = (Radius * angular_length) / (TileSizeUU * sqrt(3))
+
+  const float Phi = (1.f + FMath::Sqrt(5.f)) * 0.5f;
+  const float IcoEdgeAngle = 2.f * FMath::Atan(1.f / Phi); // ~1.1071 rad
+  const float IcoEdgeWorld = Radius * IcoEdgeAngle;
+  const float CenterToCenter =
+      FMath::Max(1.f, TileSizeUU) * 1.7320508075688772f; // sqrt(3)
+
+  int32 Freq = FMath::Max(1, FMath::RoundToInt(IcoEdgeWorld / CenterToCenter));
+  return Freq;

+void UGoldbergGrid::BuildHexMapping() {
+  HexToCellMap.Empty();
+  CellToHexMap.Empty();
+
+  if (Cells.Num() == 0) {
+    return;
+  }
+
+  // ===== IMPROVED HEX MAPPING =====
+  // Key insight: On a Goldberg polyhedron, hex coordinates must preserve
+  // adjacency ΓÇö if two cells are neighbors on the sphere, their hex coords
+  // must be hex-neighbors. Otherwise orbs "teleport" when crossing seam
+  // boundaries.
+  //
+  // Algorithm:
+  // 1. For each cell, its neighbors are already sorted in angular order.
+  // 2. BFS from a starting cell, assigning hex coordinates.
+  // 3. When we visit a neighbor, we need to determine which hex direction
+  //    to use. We do this by finding a consistent alignment: when cell A
+  //    already has a hex coordinate and we process its neighbor B at slot i,
+  //    we also check what slot A occupies in B's neighbor list, and verify
+  //    the directions are reciprocal (i.e., direction d and d+3 mod 6).
+  // 4. If there's a collision or inconsistency, we skip that neighbor
+  //    (it will be reached via another BFS path).
+
+  // Find a good starting cell (non-pentagon, near "top" of sphere)
+  int32 StartCell = INDEX_NONE;
+  float BestZ = -FLT_MAX;
+  for (int32 i = 0; i < Cells.Num(); ++i) {
+    if (Cells[i].bPentagon)
+      continue;
+    FVector Dir = (Cells[i].Position - CachedOrigin).GetSafeNormal();
+    if (Dir.Z > BestZ) {
+      BestZ = Dir.Z;
+      StartCell = i;
+    }
+  }
+
+  if (StartCell == INDEX_NONE) {
+    UE_LOG(LogTemp, Warning,
+           TEXT("GoldbergGrid: No valid starting cell found for hex mapping"));
+    return;
+  }
+
+  // Per-cell: which hex direction slot (0-5) does each neighbor occupy?
+  // CellDirMap[cellIdx][neighborSlot] = assigned hex direction (or -1)
+  // We'll compute this globally for consistency.
+
+  // For each cell, store the mapping: neighborCellIdx -> which slot it is in
+  // the neighbor list
+  TMap<uint64, int32> EdgeToSlot; // key = Pack(cellA, cellB), value = slot of B
+                                  // in A's neighbor list
+  auto PackEdge = [](int32 A, int32 B) -> uint64 {
+    return ((uint64)A << 32) | (uint64)(uint32)B;
+  };
+
+  for (int32 ci = 0; ci < Cells.Num(); ++ci) {
+    const FGoldbergCell &Cell = Cells[ci];
+    for (int32 ni = 0; ni < Cell.Neighbors.Num(); ++ni) {
+      EdgeToSlot.Add(PackEdge(ci, Cell.Neighbors[ni]), ni);
+    }
+  }
+
+  // Hex direction offsets (axial coordinates)
+  static const int32 HexDirQ[6] = {+1, +1, 0, -1, -1, 0};
+  static const int32 HexDirR[6] = {0, -1, -1, 0, +1, +1};
+
+  // BFS queue: (CellIndex, HexQ, HexR, DirectionOffset)
+  // DirectionOffset: when we arrive at a cell, we know what hex direction
+  // was used. We use the first cell's angular sort as the reference.
+  struct FBFSEntry {

+    int32 HexR;

+                     // to align with hex directions
